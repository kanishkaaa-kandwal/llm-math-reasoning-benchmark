 -- Table to track AI Math Proof Evaluations
CREATE TABLE math_evaluations (
    eval_id INTEGER PRIMARY KEY,
    topic VARCHAR(50),
    ai_model VARCHAR(30),
    status VARCHAR(20),
    error_category VARCHAR(100)
);
